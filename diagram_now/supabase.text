create table public.exam_rooms (
  id uuid not null default gen_random_uuid (),
  room_name character varying(100) not null,
  exam_date date null,
  exam_time character varying(50) null,
  course_code character varying(50) null,
  course_name character varying(200) null,
  section character varying(50) null,
  created_by uuid not null,
  created_by_name text null,
  created_at timestamp with time zone null default CURRENT_TIMESTAMP,
  updated_at timestamp with time zone null default CURRENT_TIMESTAMP,
  capacity integer null,
  is_active boolean null default true,
  start_time time without time zone null,
  end_time time without time zone null,
  constraint exam_rooms_pkey primary key (id)
) TABLESPACE pg_default;

create index IF not exists idx_exam_rooms_created_at on public.exam_rooms using btree (created_at) TABLESPACE pg_default;

create index IF not exists idx_exam_rooms_created_by on public.exam_rooms using btree (created_by) TABLESPACE pg_default;

create index IF not exists idx_exam_rooms_room_name on public.exam_rooms using btree (room_name) TABLESPACE pg_default;

create index IF not exists idx_exam_rooms_name on public.exam_rooms using btree (room_name) TABLESPACE pg_default;

create index IF not exists idx_exam_rooms_exam_date on public.exam_rooms using btree (exam_date) TABLESPACE pg_default;

create table public.profiles (
  id uuid not null default gen_random_uuid (),
  user_id uuid not null,
  full_name text null,
  student_id text null,
  role text null default 'student'::text,
  face_images jsonb null,
  created_at timestamp with time zone not null default now(),
  updated_at timestamp with time zone not null default now(),
  constraint profiles_pkey primary key (id),
  constraint profiles_user_id_key unique (user_id),
  constraint profiles_user_id_fkey foreign KEY (user_id) references auth.users (id) on delete CASCADE
) TABLESPACE pg_default;

create trigger update_profiles_updated_at BEFORE
update on profiles for EACH row
execute FUNCTION update_updated_at_column ();


create table public.qr_authentication (
  id uuid not null default gen_random_uuid (),
  user_id uuid not null,
  ip inet not null,
  status text not null default 'pending'::text,
  authenticated_at timestamp with time zone null,
  expires_at timestamp with time zone null default (now() + '00:01:00'::interval),
  constraint qr_authentication_pkey primary key (id),
  constraint qr_authentication_user_id_fkey foreign KEY (user_id) references auth.users (id) on delete CASCADE,
  constraint qr_authentication_status_check check (
    (
      status = any (
        array[
          'pending'::text,
          'authenticated'::text,
          'failed'::text,
          'expired'::text
        ]
      )
    )
  ),
  constraint valid_ip_address check ((ip is not null))
) TABLESPACE pg_default;

create index IF not exists idx_qr_authentication_user_id on public.qr_authentication using btree (user_id) TABLESPACE pg_default;

create index IF not exists idx_qr_authentication_ip on public.qr_authentication using btree (ip) TABLESPACE pg_default;

create index IF not exists idx_qr_authentication_status on public.qr_authentication using btree (status) TABLESPACE pg_default;

create index IF not exists idx_qr_authentication_authenticated_at on public.qr_authentication using btree (authenticated_at) TABLESPACE pg_default;

create index IF not exists idx_qr_authentication_expires_at on public.qr_authentication using btree (expires_at) TABLESPACE pg_default;

create index IF not exists idx_qr_authentication_user_status on public.qr_authentication using btree (user_id, status) TABLESPACE pg_default;

create index IF not exists idx_qr_authentication_user_ip on public.qr_authentication using btree (user_id, ip) TABLESPACE pg_default;

  updated_at timestamp with time zone null default CURRENT_TIMESTAMP,
  constraint room_seat_layouts_pkey primary key (id),
  constraint room_seat_layouts_columns_check check (
